import { createBrowserClient } from "@supabase/ssr";

let supabaseInstance: ReturnType<typeof createBrowserClient> | null = null;

export function createClient() {
  const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const supabaseAnonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;

  if (!supabaseUrl || !supabaseAnonKey) {
    throw new Error(
      "Faltan las variables NEXT_PUBLIC_SUPABASE_URL o NEXT_PUBLIC_SUPABASE_ANON_KEY."
    );
  }

  if (!supabaseInstance) {
    supabaseInstance = createBrowserClient(
      supabaseUrl,
      supabaseAnonKey
    );
  }

  return supabaseInstance;
}

import "server-only";
import { createClient } from "@supabase/supabase-js";

export function createAdminClient() {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const serviceKey = process.env.SUPABASE_SERVICE_ROLE_KEY;

  if (!url || !serviceKey) {
    throw new Error("Falta configurar la conexiÃ³n privada de Supabase.");
  }

  return createClient(url, serviceKey, {
    auth: {
      autoRefreshToken: [OCULTO]
      persistSession: false,
    },
  });
}

import { NextResponse } from "next/server";
import { createAdminClient } from "@/lib/supabase-admin";
import bcrypt from "bcryptjs";

export async function POST(request: Request) {
  try {
    const { cedula, codigo, pin } = await request.json();

    if (
      typeof cedula !== "string" ||
      typeof codigo !== "string" ||
      typeof pin !== "string" ||
      !/^\d{10}$/.test(cedula) ||
      !/^\d{6}$/.test(codigo) ||
      !/^\d{4}$/.test(pin)
    ) {
      return NextResponse.json(
        { error: "Los datos ingresados no son vÃ¡lidos." },
        { status: 400 }
      );
    }

    const supabase = createAdminClient();

    const { data: socio, error: socioError } = await supabase
      .from("socios")
      .select("id, estado")
      .eq("cedula", cedula)
      .maybeSingle();

    if (socioError || !socio || socio.estado !== "activo") {
      return NextResponse.json(
        { error: "No se pudo verificar tu registro." },
        { status: 400 }
      );
    }

    const { data: recuperacion, error: codigoError } = await supabase
      .from("recuperacion_pin")
      .select("id, codigo_hash, intentos, expira_at")
      .eq("socio_id", socio.id)
      .eq("usado", false)
      .order("creado_at", { ascending: false })
      .limit(1)
      .maybeSingle();

    if (
      codigoError ||
      !recuperacion ||
      new Date(recuperacion.expira_at).getTime() <= Date.now() ||
      recuperacion.intentos >= 5
    ) {
      return NextResponse.json(
        { error: "CÃ³digo invÃ¡lido o vencido. Solicita uno nuevo al administrador." },
        { status: 400 }
      );
    }

    const codigoCorrecto = await bcrypt.compare(
      codigo,
      recuperacion.codigo_hash
    );

    if (!codigoCorrecto) {
      const intentos = recuperacion.intentos + 1;

      await supabase
        .from("recuperacion_pin")
        .update({
          intentos,
          usado: intentos >= 5,
        })
        .eq("id", recuperacion.id)
        .eq("usado", false);

      return NextResponse.json(
        { error: "CÃ³digo incorrecto." },
        { status: 400 }
      );
    }

    const pinHash = await bcrypt.hash(pin, 12);

    const { data: actualizado, error: actualizarError } = await supabase
      .from("socios")
      .update({
        pin_hash: pinHash,
        pin_configurado: true,
        pin_intentos: 0,
        pin_bloqueado_hasta: null,
        pin_actualizado_at: new Date().toISOString(),
      })
      .eq("id", socio.id)
      .eq("estado", "activo")
      .select("id")
      .maybeSingle();

    if (actualizarError || !actualizado) {
      return NextResponse.json(
        { error: "No se pudo guardar el nuevo PIN." },
        { status: 500 }
      );
    }

    const { error: consumirError } = await supabase
      .from("recuperacion_pin")
      .update({ usado: true })
      .eq("id", recuperacion.id)
      .eq("usado", false);

    if (consumirError) {
      return NextResponse.json(
        { error: "El PIN fue actualizado, pero no se pudo cerrar el cÃ³digo. Contacta al administrador." },
        { status: 500 }
      );
    }

    return NextResponse.json({ correcto: true });
  } catch {
    return NextResponse.json(
      { error: "No se pudo configurar el PIN." },
      { status: 500 }
    );
  }
}
